2. You’ve discovered it!

a² + b² = c²

Pythagoras’ Theorem: in any right-angled triangle, the area of the square on the longest side (the hypotenuse) equals the sum of the areas of the squares on the other two sides.

1Create a Google Sheet and add the script

  1. Create a new Google Sheet (e.g. “Triangles & Squares – Class Results”).
  2. Click Extensions → Apps Script.
  3. Delete any existing code and paste the script below.
function safe(v){v=String(v==null?'':v);return /^[=+\-@]/.test(v)?"'"+v:v}
function doPost(e){
  try{
    var d=JSON.parse(e.postData.contents),ss=SpreadsheetApp.getActiveSpreadsheet();
    var cfg={
      observation:['Observations',['Time','Student name','What I notice','Triangles saved (a, b, c; a2, b2, c2)']],
      practice:['Practice results',['Time','Student name','Questions correct','Out of','Unsuccessful attempts (total)','Total working time','Unsuccessful: step 1 (squaring)','Unsuccessful: step 2 (+ or -)','Unsuccessful: step 3 (square root)','Details per question']],
      test:['Test',['Time','Student name','Message']]};
    var c=cfg[d.type]; if(!c) throw new Error('Unknown type');
    var sh=ss.getSheetByName(c[0])||ss.insertSheet(c[0]);
    if(sh.getLastRow()===0){sh.appendRow(c[1]);sh.setFrozenRows(1);}
    var r={
      observation:[safe(d.studentName),safe(d.observation),safe(d.table)],
      practice:[safe(d.studentName),d.correct,d.total,d.unsuccessful,safe(d.time),d.step1,d.step2,d.step3,safe(d.detail)],
      test:[safe(d.studentName),'Connection OK']};
    sh.appendRow([new Date()].concat(r[d.type]));
    return ContentService.createTextOutput('ok');
  }catch(err){return ContentService.createTextOutput('error: '+err);}
}
function doGet(){return ContentService.createTextOutput('Triangles and Squares web app is active.');}
